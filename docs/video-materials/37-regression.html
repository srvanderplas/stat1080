library(tidyr)
library(ggplot2)
library(HistData)
data("GaltonFamilies")
set.seed(1983)
galton <- as.data.frame(GaltonFamilies)

galton_sample <- GaltonFamilies |>
  filter(gender == "male") |>
  group_by(family) |>
  # Get a single, randomly selected male child
  slice_sample(n = 1) |> 
  ungroup() |>
  select(
    father,
    # rename to son
    son = childHeight 
  )

Linear regression improves precision

B <- 1000
N <- 50


results <- galton_sample |>
  rep_slice_sample(reps = B, n = N, replace = TRUE) |>
  summarise(
    # Conditional average with stratification
    ca = mean(son[round(father)==72]), 
    # Linear regression prediction
    lm = predict(lm(son ~ father), newdata = data.frame(father = 72)))

sd(results$ca, na.rm = T)
[1] 1.078234
sd(results$lm, na.rm = T)
[1] 0.5034568

Bivariate normal distribution justification

galton_sample |>
  mutate(father_z = (father-mean(father))/sd(father), 
         father_z_round = round(father_z), 
         son_z = (son - mean(son))/sd(son)) |>
  # Remove outliers
  filter(father_z_round <=2, father_z_round >= -2) |> 
  ggplot() + stat_qq(aes(sample = son_z)) + 
  facet_wrap(~father_z_round, ncol = 5) +
  geom_abline(slope = 1) + 
  coord_fixed() + 
  theme_bw()
