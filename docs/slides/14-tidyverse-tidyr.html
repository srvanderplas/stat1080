col_names <- c("Year.Month",
               paste("Week",
                     rep(1:5, each = 2),
                     rep(c("Date", "Value"), times = 5), sep = "."))
library(magrittr)

gas_prices_raw <- gas_prices_html |>
  set_names(col_names) |>
  filter(Year.Month != "Year-Month") |>
  filter(Year.Month != "")
